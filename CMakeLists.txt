# dasDuckDB builds two ways from a single CMakeLists (same pattern as dasImguiImplot):
#  - included from a daslang superbuild (the ADD_MODULE_* macros exist): the module
#    links statically into the host app.
#  - configured directly (top-level project, DASLANG_DIR set): the standalone
#    .shared_module build CI, the daspkg index, and -load_module consumers run on.
#
# DuckDB itself is NOT vendored: the official prebuilt libduckdb for the host
# platform (pinned URL + SHA256) is downloaded into duckdb/ at configure time.
# Prebuilt, NOT the source amalgamation: the amalgamation ships without the
# statically-bundled extensions (core_functions, json), so sum()/min()/max()/
# json_extract() are missing from it — the prebuilt libraries include them.
# The shared library is copied next to dasModuleDuckDB.shared_module; daslang's
# module loader searches the loaded module's own directory for dependents
# (LOAD_LIBRARY_SEARCH_DLL_LOAD_DIR on Windows, $ORIGIN rpath on POSIX).
IF(NOT COMMAND ADD_MODULE_CPP)
    cmake_minimum_required(VERSION 3.18)
    project(dasDuckDB CXX C)
ENDIF()

IF((NOT DAS_DUCKDB_INCLUDED) AND ((NOT ${DAS_DUCKDB_DISABLED}) OR (NOT DEFINED DAS_DUCKDB_DISABLED)))
SET(DAS_DUCKDB_INCLUDED TRUE)
MESSAGE(STATUS "dasDuckDB module included.")

SET(DAS_DUCKDB_DIR ${CMAKE_CURRENT_LIST_DIR})

# ===== DuckDB prebuilt library (downloaded, pinned) =====
SET(DUCKDB_VERSION "1.5.4")
IF(WIN32)
    IF(CMAKE_SYSTEM_PROCESSOR MATCHES "ARM64|aarch64")
        SET(DUCKDB_PLATFORM "windows-arm64")
        SET(DUCKDB_LIB_SHA256 "8056a9fc58a7c0ced9ebd5eb804e1bd2caf0e1ffb6711c8bd0c3ac0558a7da7f")
    ELSE()
        SET(DUCKDB_PLATFORM "windows-amd64")
        SET(DUCKDB_LIB_SHA256 "74e73afd3b010c6f310e14a961fff679f876952bc196f82584b0e2e76d11a91f")
    ENDIF()
    SET(DUCKDB_SHARED_LIB "duckdb.dll")
    SET(DUCKDB_LINK_LIB "duckdb.lib")
ELSEIF(APPLE)
    SET(DUCKDB_PLATFORM "osx-universal")
    SET(DUCKDB_LIB_SHA256 "3f3c52970ad1407ec5037062e1a5e575b24bd5b993c889f89fe5876eff47782c")
    SET(DUCKDB_SHARED_LIB "libduckdb.dylib")
    SET(DUCKDB_LINK_LIB "libduckdb.dylib")
ELSE()
    IF(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
        SET(DUCKDB_PLATFORM "linux-arm64")
        SET(DUCKDB_LIB_SHA256 "7e154fd14d5b7afd7c9f36071000bf4badd3fa57a125cf7dc596f2c5ad82c1b2")
    ELSE()
        SET(DUCKDB_PLATFORM "linux-amd64")
        SET(DUCKDB_LIB_SHA256 "838d98a85e697bab9935010c88a8c67d3312ccedcab4cb4a0ba01da65113bb70")
    ENDIF()
    SET(DUCKDB_SHARED_LIB "libduckdb.so")
    SET(DUCKDB_LINK_LIB "libduckdb.so")
ENDIF()

SET(DUCKDB_DIR ${DAS_DUCKDB_DIR}/duckdb)
IF(NOT EXISTS "${DUCKDB_DIR}/${DUCKDB_SHARED_LIB}")
    SET(_duckdb_zip ${CMAKE_BINARY_DIR}/libduckdb-${DUCKDB_PLATFORM}-${DUCKDB_VERSION}.zip)
    MESSAGE(STATUS "Downloading DuckDB ${DUCKDB_VERSION} (${DUCKDB_PLATFORM}) prebuilt library...")
    FILE(DOWNLOAD
        "https://github.com/duckdb/duckdb/releases/download/v${DUCKDB_VERSION}/libduckdb-${DUCKDB_PLATFORM}.zip"
        ${_duckdb_zip}
        EXPECTED_HASH SHA256=${DUCKDB_LIB_SHA256}
        SHOW_PROGRESS)
    FILE(ARCHIVE_EXTRACT INPUT ${_duckdb_zip} DESTINATION ${DUCKDB_DIR})
ENDIF()

# binding source files (generated by bind/bind_duckdb.das + hand-written main.cpp).
# The func_*.cpp chunk count varies with the bound-function count, so glob them —
# always `rm src/dasDUCKDB.func_*.cpp` before regenerating, then re-run CMake configure.
file(GLOB DAS_DUCKDB_FUNC_SRC "${DAS_DUCKDB_DIR}/src/dasDUCKDB.func_*.cpp")
SET(DAS_DUCKDB_BIND_SRC
    ${DAS_DUCKDB_DIR}/src/dasDUCKDB.cpp
    ${DAS_DUCKDB_DIR}/src/dasDUCKDB.main.cpp
    ${DAS_DUCKDB_DIR}/src/dasDUCKDB.userfn.cpp
    ${DAS_DUCKDB_FUNC_SRC}
)

IF(COMMAND ADD_MODULE_CPP)
    # ===== daslang superbuild: static-embed =====
    # `require duckdb/<name>` -> daslib/<name>.das; the glob IS the
    # .das_module register_native_path list (file stem == module name throughout).
    # NOTE: the host app must ship ${DUCKDB_SHARED_LIB} next to its executable.
    ADD_MODULE_CPP(dasDUCKDB)
    FILE(GLOB _dasDuckDB_das_files "${DAS_DUCKDB_DIR}/daslib/*.das")
    FOREACH(_f ${_dasDuckDB_das_files})
        GET_FILENAME_COMPONENT(_n "${_f}" NAME_WE)
        ADD_MODULE_DAS(duckdb daslib ${_n})
    ENDFOREACH()

    ADD_DAS_STATIC_MODULE_LIB(libDasModuleDuckDB ${DAS_DUCKDB_BIND_SRC})
    TARGET_INCLUDE_DIRECTORIES(libDasModuleDuckDB PRIVATE
        ${DUCKDB_DIR}
        ${DAS_DUCKDB_DIR}/src
    )
    TARGET_COMPILE_FEATURES(libDasModuleDuckDB PRIVATE cxx_std_17)
    TARGET_LINK_LIBRARIES(libDasModuleDuckDB PUBLIC ${DUCKDB_DIR}/${DUCKDB_LINK_LIB})
ELSE()
    # ===== standalone: dasModuleDuckDB.shared_module =====
    if(NOT DEFINED DASLANG_DIR)
        message(FATAL_ERROR "DASLANG_DIR must be set to the daslang SDK root")
    endif()

    add_library(dasModuleDuckDB SHARED ${DAS_DUCKDB_BIND_SRC})
    target_compile_features(dasModuleDuckDB PRIVATE cxx_std_17)
    target_compile_definitions(dasModuleDuckDB PRIVATE
        DAS_MOD_EXPORTS DAS_ENABLE_DLL=1
        $<$<CONFIG:Release>:DAS_NO_ASSERTIONS>
        $<$<CONFIG:MinSizeRel>:DAS_NO_ASSERTIONS>
        $<$<CONFIG:RelWithDebInfo>:DAS_NO_ASSERTIONS>
    )
    target_include_directories(dasModuleDuckDB PRIVATE
        ${DASLANG_DIR}/include
        ${DUCKDB_DIR}
        ${DAS_DUCKDB_DIR}/src
    )
    target_link_libraries(dasModuleDuckDB PRIVATE ${DUCKDB_DIR}/${DUCKDB_LINK_LIB})
    if(WIN32)
        if(EXISTS "${DASLANG_DIR}/lib/Release/libDaScriptDyn.lib")
            target_link_libraries(dasModuleDuckDB PRIVATE
                ${DASLANG_DIR}/lib/Release/libDaScriptDyn.lib
                ${DASLANG_DIR}/lib/Release/libDaScriptDyn_runtime.lib
            )
        else()
            target_link_libraries(dasModuleDuckDB PRIVATE
                ${DASLANG_DIR}/lib/libDaScriptDyn.lib
                ${DASLANG_DIR}/lib/libDaScriptDyn_runtime.lib
            )
        endif()
    elseif(APPLE)
        target_link_libraries(dasModuleDuckDB PRIVATE
            ${DASLANG_DIR}/lib/liblibDaScriptDyn.dylib
            ${DASLANG_DIR}/lib/liblibDaScriptDyn_runtime.dylib
        )
    else()
        target_link_libraries(dasModuleDuckDB PRIVATE
            ${DASLANG_DIR}/lib/liblibDaScriptDyn.so
            ${DASLANG_DIR}/lib/liblibDaScriptDyn_runtime.so
        )
    endif()
    if(UNIX)
        find_package(Threads REQUIRED)
        target_link_libraries(dasModuleDuckDB PRIVATE Threads::Threads)
        if(NOT APPLE)
            target_link_libraries(dasModuleDuckDB PRIVATE dl m)
        endif()
    endif()
    if(MSVC)
        target_compile_options(dasModuleDuckDB PRIVATE /wd4005 /bigobj)
    else()
        # daslang runtime libs are built -fno-rtti for optimized configs; match the ABI
        # or the .shared_module mislinks.
        target_compile_options(dasModuleDuckDB PRIVATE
            $<$<CONFIG:Release>:-fno-rtti>
            $<$<CONFIG:MinSizeRel>:-fno-rtti>
            $<$<CONFIG:RelWithDebInfo>:-fno-rtti>)
    endif()
    set_target_properties(dasModuleDuckDB PROPERTIES
        SUFFIX ".shared_module"
        PREFIX ""
        RUNTIME_OUTPUT_DIRECTORY_RELEASE "${DAS_DUCKDB_DIR}"
        LIBRARY_OUTPUT_DIRECTORY "${DAS_DUCKDB_DIR}"
    )
    if(APPLE)
        # libduckdb.dylib's install name is @rpath-based; resolve it from the module's own dir
        set_target_properties(dasModuleDuckDB PROPERTIES
            BUILD_WITH_INSTALL_RPATH TRUE
            INSTALL_RPATH "@loader_path")
    elseif(UNIX)
        set_target_properties(dasModuleDuckDB PROPERTIES
            BUILD_WITH_INSTALL_RPATH TRUE
            INSTALL_RPATH "$ORIGIN")
    endif()
    # the runtime library rides next to the .shared_module (the module loader
    # searches the loaded module's own directory for its dependents)
    add_custom_command(TARGET dasModuleDuckDB POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "${DUCKDB_DIR}/${DUCKDB_SHARED_LIB}" "${DAS_DUCKDB_DIR}/${DUCKDB_SHARED_LIB}")
ENDIF() # superbuild / standalone

ENDIF() # DAS_DUCKDB_INCLUDED guard
