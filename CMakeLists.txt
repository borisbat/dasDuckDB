# dasDuckDB builds two ways from a single CMakeLists (same pattern as dasImguiImplot):
#  - included from a daslang superbuild (the ADD_MODULE_* macros exist): the module
#    links statically into the host app.
#  - configured directly (top-level project, DASLANG_DIR set): the standalone
#    .shared_module build CI, the daspkg index, and -load_module consumers run on.
#
# DuckDB itself is NOT vendored: the release amalgamation (duckdb.cpp + duckdb.h,
# pinned URL + SHA256) is downloaded into duckdb/ at configure time and compiled
# straight into the module — self-contained, no runtime DLL to distribute.
IF(NOT COMMAND ADD_MODULE_CPP)
    cmake_minimum_required(VERSION 3.18)
    project(dasDuckDB CXX C)
ENDIF()

IF((NOT DAS_DUCKDB_INCLUDED) AND ((NOT ${DAS_DUCKDB_DISABLED}) OR (NOT DEFINED DAS_DUCKDB_DISABLED)))
SET(DAS_DUCKDB_INCLUDED TRUE)
MESSAGE(STATUS "dasDuckDB module included.")

SET(DAS_DUCKDB_DIR ${CMAKE_CURRENT_LIST_DIR})

# ===== DuckDB amalgamation (downloaded, pinned) =====
SET(DUCKDB_VERSION "1.5.4")
SET(DUCKDB_SRC_SHA256 "28ad9fd34607dff7ee27c2cb8e5ef77b51bb7c1a214ac3e4267dcaaeeb752b62")
SET(DUCKDB_SRC_DIR ${DAS_DUCKDB_DIR}/duckdb)
IF(NOT EXISTS "${DUCKDB_SRC_DIR}/duckdb.cpp")
    SET(_duckdb_zip ${CMAKE_BINARY_DIR}/libduckdb-src-${DUCKDB_VERSION}.zip)
    MESSAGE(STATUS "Downloading DuckDB ${DUCKDB_VERSION} amalgamation...")
    FILE(DOWNLOAD
        "https://github.com/duckdb/duckdb/releases/download/v${DUCKDB_VERSION}/libduckdb-src.zip"
        ${_duckdb_zip}
        EXPECTED_HASH SHA256=${DUCKDB_SRC_SHA256}
        SHOW_PROGRESS)
    FILE(ARCHIVE_EXTRACT INPUT ${_duckdb_zip} DESTINATION ${DUCKDB_SRC_DIR})
ENDIF()

SET(DAS_DUCKDB_SRC ${DUCKDB_SRC_DIR}/duckdb.cpp)
# vendor amalgamation is not ours to fix: silence its warnings; /bigobj for the 25MB TU
IF(MSVC)
    SET_SOURCE_FILES_PROPERTIES(${DAS_DUCKDB_SRC} PROPERTIES COMPILE_OPTIONS "/w;/bigobj")
ELSE()
    SET_SOURCE_FILES_PROPERTIES(${DAS_DUCKDB_SRC} PROPERTIES COMPILE_OPTIONS "-w")
ENDIF()

# binding source files (generated by bind/bind_duckdb.das + hand-written main.cpp).
# The func_*.cpp chunk count varies with the bound-function count, so glob them —
# always `rm src/dasDUCKDB.func_*.cpp` before regenerating, then re-run CMake configure.
file(GLOB DAS_DUCKDB_FUNC_SRC "${DAS_DUCKDB_DIR}/src/dasDUCKDB.func_*.cpp")
SET(DAS_DUCKDB_BIND_SRC
    ${DAS_DUCKDB_DIR}/src/dasDUCKDB.cpp
    ${DAS_DUCKDB_DIR}/src/dasDUCKDB.main.cpp
    ${DAS_DUCKDB_FUNC_SRC}
)

IF(COMMAND ADD_MODULE_CPP)
    # ===== daslang superbuild: static-embed =====
    # `require duckdb/<name>` -> daslib/<name>.das; the glob IS the
    # .das_module register_native_path list (file stem == module name throughout).
    ADD_MODULE_CPP(dasDUCKDB)
    FILE(GLOB _dasDuckDB_das_files "${DAS_DUCKDB_DIR}/daslib/*.das")
    FOREACH(_f ${_dasDuckDB_das_files})
        GET_FILENAME_COMPONENT(_n "${_f}" NAME_WE)
        ADD_MODULE_DAS(duckdb daslib ${_n})
    ENDFOREACH()

    ADD_DAS_STATIC_MODULE_LIB(libDasModuleDuckDB ${DAS_DUCKDB_BIND_SRC} ${DAS_DUCKDB_SRC})
    TARGET_INCLUDE_DIRECTORIES(libDasModuleDuckDB PRIVATE
        ${DUCKDB_SRC_DIR}
        ${DAS_DUCKDB_DIR}/src
    )
    TARGET_COMPILE_DEFINITIONS(libDasModuleDuckDB PRIVATE DUCKDB_STATIC_BUILD=1
        $<$<PLATFORM_ID:Windows>:WIN32_LEAN_AND_MEAN>
        $<$<PLATFORM_ID:Windows>:NOMINMAX>)
    TARGET_COMPILE_FEATURES(libDasModuleDuckDB PRIVATE cxx_std_17)
    IF(UNIX)
        FIND_PACKAGE(Threads REQUIRED)
        TARGET_LINK_LIBRARIES(libDasModuleDuckDB PRIVATE Threads::Threads)
        IF(NOT APPLE)
            TARGET_LINK_LIBRARIES(libDasModuleDuckDB PRIVATE dl m)
        ENDIF()
    ELSEIF(WIN32)
        # DuckDB's Windows file-lock diagnostics use the Restart Manager API
        TARGET_LINK_LIBRARIES(libDasModuleDuckDB PRIVATE rstrtmgr)
    ENDIF()
ELSE()
    # ===== standalone: dasModuleDuckDB.shared_module =====
    if(NOT DEFINED DASLANG_DIR)
        message(FATAL_ERROR "DASLANG_DIR must be set to the daslang SDK root")
    endif()

    add_library(dasModuleDuckDB SHARED ${DAS_DUCKDB_BIND_SRC} ${DAS_DUCKDB_SRC})
    target_compile_features(dasModuleDuckDB PRIVATE cxx_std_17)
    target_compile_definitions(dasModuleDuckDB PRIVATE
        DAS_MOD_EXPORTS DAS_ENABLE_DLL=1
        DUCKDB_STATIC_BUILD=1
        # duckdb.cpp uses `interface` as an identifier; ole2.h (pulled by fat windows.h)
        # macro-defines it. NOMINMAX for the usual min/max collisions.
        $<$<PLATFORM_ID:Windows>:WIN32_LEAN_AND_MEAN>
        $<$<PLATFORM_ID:Windows>:NOMINMAX>
        $<$<CONFIG:Release>:DAS_NO_ASSERTIONS>
        $<$<CONFIG:MinSizeRel>:DAS_NO_ASSERTIONS>
        $<$<CONFIG:RelWithDebInfo>:DAS_NO_ASSERTIONS>
    )
    target_include_directories(dasModuleDuckDB PRIVATE
        ${DASLANG_DIR}/include
        ${DUCKDB_SRC_DIR}
        ${DAS_DUCKDB_DIR}/src
    )
    if(WIN32)
        # DuckDB's Windows file-lock diagnostics use the Restart Manager API
        target_link_libraries(dasModuleDuckDB PRIVATE rstrtmgr)
        if(EXISTS "${DASLANG_DIR}/lib/Release/libDaScriptDyn.lib")
            target_link_libraries(dasModuleDuckDB PRIVATE
                ${DASLANG_DIR}/lib/Release/libDaScriptDyn.lib
                ${DASLANG_DIR}/lib/Release/libDaScriptDyn_runtime.lib
            )
        else()
            target_link_libraries(dasModuleDuckDB PRIVATE
                ${DASLANG_DIR}/lib/libDaScriptDyn.lib
                ${DASLANG_DIR}/lib/libDaScriptDyn_runtime.lib
            )
        endif()
    elseif(APPLE)
        target_link_libraries(dasModuleDuckDB PRIVATE
            ${DASLANG_DIR}/lib/liblibDaScriptDyn.dylib
            ${DASLANG_DIR}/lib/liblibDaScriptDyn_runtime.dylib
        )
    else()
        target_link_libraries(dasModuleDuckDB PRIVATE
            ${DASLANG_DIR}/lib/liblibDaScriptDyn.so
            ${DASLANG_DIR}/lib/liblibDaScriptDyn_runtime.so
        )
    endif()
    if(UNIX)
        find_package(Threads REQUIRED)
        target_link_libraries(dasModuleDuckDB PRIVATE Threads::Threads)
        if(NOT APPLE)
            target_link_libraries(dasModuleDuckDB PRIVATE dl m)
        endif()
    endif()
    if(MSVC)
        target_compile_options(dasModuleDuckDB PRIVATE /wd4005 /bigobj)
    else()
        # daslang runtime libs are built -fno-rtti for optimized configs; match the ABI
        # or the .shared_module mislinks. DuckDB's amalgamation needs rtti, so only the
        # binding TUs get the flag via per-file options below.
        set_source_files_properties(${DAS_DUCKDB_BIND_SRC} PROPERTIES COMPILE_OPTIONS
            "$<$<CONFIG:Release>:-fno-rtti>;$<$<CONFIG:MinSizeRel>:-fno-rtti>;$<$<CONFIG:RelWithDebInfo>:-fno-rtti>")
    endif()
    set_target_properties(dasModuleDuckDB PROPERTIES
        SUFFIX ".shared_module"
        PREFIX ""
        RUNTIME_OUTPUT_DIRECTORY_RELEASE "${DAS_DUCKDB_DIR}"
        LIBRARY_OUTPUT_DIRECTORY "${DAS_DUCKDB_DIR}"
    )
ENDIF() # superbuild / standalone

ENDIF() # DAS_DUCKDB_INCLUDED guard
